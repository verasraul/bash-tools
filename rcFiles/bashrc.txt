function get_ssh_key_files() {
  local list=
  for keyfile in $(/usr/bin/egrep -d skip -c 'BEGIN.+PRIVATE KEY' ~/.ssh/* | grep :1 | cut -d: -f1); do
    ssh-keygen -lf ${keyfile}.pub &>/dev/null && list="${list} $(basename ${keyfile})"
  done
  echo $list
}

# set up public key using above function
export GPG_AGENT_INFO="~/.gnupg/S.gpg-agent:$(pgrep gpg-agent):1"
eval `keychain --noask --eval --agents gpg,ssh --inherit any --timeout 28800 $(get_ssh_key_files)`
